\section{Wykaz symboli, oznaczeń i~akronimów}

\begin{longtable}{@{}>{\bfseries}p{0.15\textwidth}p{0.8\textwidth}@{}}
  AI & (ang. \emph{Artificial Intelligence}) -- sztuczna inteligencja \\
  API & (ang. \emph{Application Programming Interface}) -- interfejs programistyczny aplikacji \\
  CI & (ang. \emph{Continuous Integration}) -- ciągła integracja; automatyczne budowanie i~testowanie każdej zmiany w~kodzie \\
  CORD & (ang. \emph{Consolidated Receipt Dataset}) -- zbiór danych paragonów do oceny parsowania ich struktury \\
  E2E & (ang. \emph{end-to-end}) -- testy całego systemu z~perspektywy użytkownika \\
  ICDAR & (ang. \emph{International Conference on Document Analysis and Recognition}) -- międzynarodowa konferencja poświęcona analizie i~rozpoznawaniu dokumentów \\
  ISO 4217 & międzynarodowa norma definiująca kody walut i~ich jednostki podrzędne \\
  JSON & (ang. \emph{JavaScript Object Notation}) -- tekstowy format wymiany danych \\
  LLM & (ang. \emph{Large Language Model}) -- duży model językowy \\
  NP & klasa złożoności obliczeniowej problemów decyzyjnych, których rozwiązanie można zweryfikować w~czasie wielomianowym \\
  OCR & (ang. \emph{Optical Character Recognition}) -- optyczne rozpoznawanie znaków \\
  ORM & (ang. \emph{Object-Relational Mapping}) -- odwzorowanie obiektowo-relacyjne \\
  RWD & (ang. \emph{Responsive Web Design}) -- projektowanie stron dopasowujących się do rozmiaru ekranu \\
  S3 & (ang. \emph{Simple Storage Service}) -- interfejs magazynu obiektów, stosowany również przez rozwiązania zgodne z~usługą Amazon S3 \\
  SROIE & (ang. \emph{Scanned Receipt OCR and Information Extraction}) -- konkurs i~zbiór danych dotyczący odczytu paragonów \\
  ZPI & Zespołowe Przedsięwzięcie Informatyczne \\
\end{longtable}
